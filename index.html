// วางโค้ดนี้ใน Google Sheet > ส่วนขยาย > Apps Script // ตั้งรหัสแอดมินที่: Project Settings (รูปเฟือง) > Script Properties > Add property ชื่อ ADMIN_KEY // ครั้งแรกหลังวางโค้ด: เลือกฟังก์ชัน getFolder ที่แถบด้านบน กด Run แล้วกดอนุญาตสิทธิ์ Google Drive const SHEET_NAME = 'ทะเบียนผู้อบรม'; const FOLDER_NAME = 'ใบขับขี่ผู้อบรม (ห้ามแชร์)'; const MAX_PHOTOS_PER_DAY = 300; const HEAD = ['เวลา','ผลสอบ','ครั้งที่สอบ','เลขที่ใบรับรอง','ประเภทรถ','ชื่อ-นามสกุล', 'เลขที่ใบขับขี่','ทะเบียนรถ','บริษัทผู้ประกอบการ','ใบขับขี่','หลักสูตร', 'คะแนนก่อนเรียน','คะแนนหลังเรียน','คะแนนเต็ม','รหัสรายการ', 'วันเวลาที่ยินยอมข้อมูลส่วนบุคคล','วันหมดอายุใบขับขี่','รหัสรูปบัตร','ลิงก์รูปบัตร']; // บังคับเก็บเป็นข้อความ (กันเลข 0 นำหน้าหาย กันถูกตีเป็นวันที่หรือสูตรในชีต) const t = v => (typeof v === 'string' && v !== '') ? "'" + v : v; const json = o => ContentService.createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); // ลิงก์ที่กดเปิดได้ในชีต (เปิดได้เฉพาะบัญชี Google ที่เป็นเจ้าของโฟลเดอร์รูป) function photoLink(id) { return (id && /^[\w-]+$/.test(String(id))) ? '=HYPERLINK("https://drive.google.com/file/d/' + id + '/view","ดูรูป")' : ''; } // รันครั้งเดียวด้วยมือ: เติมลิงก์รูปให้แถวเก่าที่มีรหัสรูปแต่ยังไม่มีลิงก์ function fillPhotoLinks() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if (!sh || sh.getLastRow() < 2) return; ensureHeader(sh); const n = sh.getLastRow() - 1; const ids = sh.getRange(2, 18, n, 1).getValues(); const links = sh.getRange(2, 19, n, 1).getValues(); let c = 0; for (let i = 0; i < n; i++) { const f = photoLink(ids[i][0]); if (f && !links[i][0]) { sh.getRange(i + 2, 19).setFormula(f); c++; } } Logger.log('เติมลิงก์รูป ' + c + ' แถว'); } function adminOk(d) { const k = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY'); return !!k && d.key === k; } function getSheet(ss) { let sh = ss.getSheetByName(SHEET_NAME); if (sh) return sh; const all = ss.getSheets(); // ถ้ามีแท็บเดียวและยังว่าง ให้ใช้แท็บนั้นเลย if (all.length === 1 && all[0].getLastRow() === 0) { all[0].setName(SHEET_NAME); return all[0]; } return ss.insertSheet(SHEET_NAME); } function ensureHeader(sh) { if (sh.getLastRow() === 0) { sh.appendRow(HEAD); return; } if (sh.getRange(1, 7).getValue() === 'รหัสพนักงาน') sh.getRange(1, 7).setValue('เลขที่ใบขับขี่'); for (let c = 16; c <= HEAD.length; c++) { if (!sh.getRange(1, c).getValue()) sh.getRange(1, c).setValue(HEAD[c - 1]); } } // โฟลเดอร์เก็บรูปบัตร อยู่ใน Drive ของเจ้าของสคริปต์ เป็นส่วนตัว (ไม่แชร์) function getFolder() { const it = DriveApp.getFoldersByName(FOLDER_NAME); return it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER_NAME); } // ผู้เรียนส่งรูปบัตร (รับเฉพาะ JPEG ขนาดจำกัด และจำกัดจำนวนต่อวัน) function savePhoto(d) { if (typeof d.data !== 'string' || d.data.length < 100 || d.data.length > 1500000) { return json({ ok: false, error: 'size' }); } let bytes; try { bytes = Utilities.base64Decode(d.data); } catch (err) { return json({ ok: false, error: 'format' }); } if (bytes.length < 4 || (bytes[0] & 255) !== 0xFF || (bytes[1] & 255) !== 0xD8) { return json({ ok: false, error: 'format' }); } const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const props = PropertiesService.getScriptProperties(); const k = 'up' + Utilities.formatDate(new Date(), 'Asia/Bangkok', 'yyyyMMdd'); const n = Number(props.getProperty(k) || 0); if (n >= MAX_PHOTOS_PER_DAY) return json({ ok: false, error: 'limit' }); props.setProperty(k, String(n + 1)); const name = 'lic-' + Date.now() + '-' + Math.floor(Math.random() * 1e6) + '.jpg'; const file = getFolder().createFile(Utilities.newBlob(bytes, 'image/jpeg', name)); return json({ ok: true, id: file.getId() }); } finally { lock.releaseLock(); } } // แอดมินขอดูรูป (ต้องมีรหัส และไฟล์ต้องอยู่ในโฟลเดอร์ของเราเท่านั้น) function getPhoto(d) { if (!adminOk(d)) return json({ ok: false }); try { const file = DriveApp.getFileById(String(d.id)); const folderId = getFolder().getId(); let inFolder = false; const ps = file.getParents(); while (ps.hasNext()) if (ps.next().getId() === folderId) inFolder = true; if (!inFolder) return json({ ok: false, error: 'notfound' }); return json({ ok: true, data: Utilities.base64Encode(file.getBlob().getBytes()) }); } catch (err) { return json({ ok: false, error: 'notfound' }); } } // ลบรูปที่ไม่มีแถวไหนอ้างถึงแล้ว (ย้ายไปถังขยะของ Drive) function purgePhotos(ids, sh) { if (!ids.length) return; const keep = {}; const last = sh.getLastRow(); if (last > 1) { sh.getRange(2, 18, last - 1, 1).getValues().forEach(r => { if (r[0]) keep[String(r[0])] = true; }); } ids.forEach(id => { if (id && !keep[id]) { try { DriveApp.getFileById(id).setTrashed(true); } catch (err) {} } }); } function doPost(e) { const d = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (d.action === 'photo') return savePhoto(d); if (d.action === 'getphoto') return getPhoto(d); // แอดมินขอรายชื่อ (ต้องมีรหัสถูกต้อง) if (d.action === 'list') { if (!adminOk(d)) return json({ ok: false }); const sh = ss.getSheetByName(SHEET_NAME); return json({ ok: true, rows: sh ? sh.getDataRange().getValues() : [] }); } // แอดมินลบรายการ (ลบตามที่เลือก หรือล้างทั้งหมด) ต้องมีรหัสถูกต้อง if (d.action === 'delete' || d.action === 'clear') { if (!adminOk(d)) return json({ ok: false }); const dlock = LockService.getScriptLock(); dlock.waitLock(10000); try { const dsh = ss.getSheetByName(SHEET_NAME); if (!dsh) return json({ ok: true, deleted: 0 }); const photoIds = []; if (d.action === 'clear') { const cnt = Math.max(dsh.getLastRow() - 1, 0); if (cnt > 0) { dsh.getRange(2, 18, cnt, 1).getValues().forEach(r => { if (r[0]) photoIds.push(String(r[0])); }); dsh.deleteRows(2, cnt); } purgePhotos(photoIds, dsh); return json({ ok: true, deleted: cnt }); } let n = 0; // ลบจากแถวล่างขึ้นบน เพื่อไม่ให้เลขแถวเลื่อน และตรวจรหัสรายการ+เวลาก่อนลบทุกครั้ง (d.items || []).slice().sort((a, b) => b.row - a.row).forEach(it => { if (it.row < 2 || it.row > dsh.getLastRow()) return; const id = String(dsh.getRange(it.row, 15).getValue() || ''); const ts = new Date(dsh.getRange(it.row, 1).getValue()).getTime(); const want = new Date(it.time).getTime(); const sameTime = (isNaN(ts) && isNaN(want)) || Math.abs(ts - want) < 2000; if (id === String(it.id || '') && sameTime) { const pid = String(dsh.getRange(it.row, 18).getValue() || ''); if (pid) photoIds.push(pid); dsh.deleteRow(it.row); n++; } }); purgePhotos(photoIds, dsh); return json({ ok: true, deleted: n }); } finally { dlock.releaseLock(); } } // บันทึกผลสอบจากหน้าเรียน const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sh = getSheet(ss); ensureHeader(sh); // กันบันทึกซ้ำ ถ้าเว็บส่งรายการเดิมมาอีกครั้ง if (d.id && sh.getLastRow() > 1 && sh.createTextFinder(String(d.id)).matchEntireCell(true).findNext()) { return ContentService.createTextOutput('ok'); } sh.appendRow([new Date(d.time), d.result, d.attempt, d.no, d.type, t(d.name), t(d.licno || d.emp), t(d.plate), t(d.company), d.license, d.course, d.pre, d.post, d.total, d.id, t(d.consentAt || ''), t(d.expiry || ''), d.photoId || '', photoLink(d.photoId)] .map(v => v === undefined ? '' : v)); return ContentService.createTextOutput('ok'); } finally { lock.releaseLock(); } } function doGet() { return ContentService.createTextOutput('Apps Script พร้อมใช้งาน'); }
